# AOFs whose parts the orders serve beyond plain comparisons: `<>`, YEAR(),
# the first field of a key of several fields; in ADT and FoxPro tables.
sql CREATE TABLE p (id INTEGER, name CHAR(10), ciname CICHAR(10), amount MONEY, d DOUBLE, born DATE, code INTEGER, n NUMERIC(6, 2))
sql CREATE INDEX id ON p (id)
sql CREATE INDEX name ON p (name)
sql CREATE INDEX ciname ON p (ciname)
sql CREATE INDEX amt ON p (amount DESC)
sql CREATE INDEX d ON p (d)
sql CREATE INDEX born ON p (born)
sql CREATE INDEX n ON p (n)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('p', 'p', 'upname', 'UPPER(name)', NULL, 2, 512, NULL)
sql INSERT INTO p VALUES (1, 'Bob', 'Bob', 10, 0.1, '1990-01-05', 1, 1.5)

# FoxPro: YEAR() through a DTOS() key; `<>`.
tabletype cdx
sql CREATE TABLE c (id INTEGER, name CHAR(10), born DATE)
sql CREATE INDEX id ON c (id)
sql CREATE INDEX name ON c (name)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('c', 'c', 'dtos', 'DTOS(born)', NULL, 2, 0, NULL)
sql INSERT INTO c VALUES (1, 'Bob', '1990-01-05')
sql INSERT INTO c VALUES (2, 'alice', '1985-03-01')
sql INSERT INTO c VALUES (3, 'Carl', '1990-07-07')
sql INSERT INTO c VALUES (4, 'Bobby', '2000-02-02')
sql INSERT INTO c VALUES (5, 'Bob', NULL)
sql INSERT INTO c VALUES (6, NULL, '1990-12-31')
