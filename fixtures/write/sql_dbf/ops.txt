# SQL on DBF tables: the statement's table type (AdsStmtSetTableType)
# decides what CREATE TABLE makes and what the other statements open.
tabletype ntx
sql CREATE TABLE ntx_t (id integer, name char(10), v varchar(12), m memo, b blob, d date, n numeric(8,2), l logical, f double, g bigint, u guid, nc nchar(4), nv nvarchar(6), nm nmemo)
sql INSERT INTO ntx_t (id, name, v, m, d, n, l, f, g) VALUES (1, 'one', 'short', 'a memo', '2020-01-02', 1.5, true, 2.25, 1234567890123)
sql INSERT INTO ntx_t (id, name) VALUES (2, 'two')
sql INSERT INTO ntx_t (id, name, v, m, nc, nv, nm) VALUES (3, 'three', 'a longer one', 'another memo value that is longer than one block of the memo file, which is sixty-four bytes or more', 'ab', 'abc', 'nmemo')
sql INSERT INTO ntx_t (id) VALUES (NULL)
sql CREATE INDEX ntx_name ON ntx_t (name)
sql CREATE UNIQUE INDEX ntx_id ON ntx_t (id)
sql CREATE INDEX ntx_d ON ntx_t (d DESC)
sql UPDATE ntx_t SET name = 'zwei', m = 'changed' WHERE id = 2
sql UPDATE ntx_t SET n = n + 1 WHERE n IS NOT NULL
sql DELETE FROM ntx_t WHERE id = 1
sql INSERT INTO ntx_t (id, name) VALUES (5, 'five')
sql INSERT INTO ntx_t (id, name) VALUES (3, 'dup')
sql CREATE INDEX ntx_two ON ntx_t (name, d)
sql EXECUTE PROCEDURE sp_CreateIndex90('ntx_t', NULL, 'ntx_ex', 'UPPER(name)', 'id > 1', 0, 512, NULL)
sql DROP INDEX ntx_t.ntx_d
sql ALTER TABLE ntx_t ADD COLUMN extra char(3)
sql INSERT INTO ntx_t (id, extra) VALUES (6, 'xyz')
sql CREATE TABLE ntx_gone (id integer, m memo)
sql CREATE INDEX ntx_gid ON ntx_gone (id)
sql DROP TABLE ntx_gone
