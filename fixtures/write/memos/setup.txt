# Tables with memo, binary and NMemo fields, four block sizes and compressed.
sql CREATE TABLE m8 (id INTEGER, m MEMO, b BLOB, n NMEMO) MEMOBLOCKSIZE 8
sql CREATE TABLE m64 (id INTEGER, m MEMO, b BLOB, n NMEMO) MEMOBLOCKSIZE 64
sql CREATE TABLE m256 (id INTEGER, m MEMO, b BLOB, n NMEMO)
sql CREATE TABLE m1024 (id INTEGER, m MEMO, b BLOB, n NMEMO) MEMOBLOCKSIZE 1024
sql CREATE TABLE c (id INTEGER, m MEMO COMPRESSED, b BLOB COMPRESSED, n NMEMO COMPRESSED)
