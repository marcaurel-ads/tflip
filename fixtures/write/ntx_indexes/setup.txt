# Clipper tables with .ntx indexes, created before any data.
tabletype ntx
sql CREATE TABLE t (id INTEGER, c CHAR(10), n NUMERIC(8, 2), d DATE, l LOGICAL, b DOUBLE(2), m MEMO, bl BLOB)
sql CREATE INDEX t_id ON t (id)
sql CREATE INDEX t_c ON t (c)
sql CREATE INDEX t_n ON t (n)
sql CREATE INDEX t_d ON t (d)
sql CREATE INDEX t_l ON t (l)
sql CREATE INDEX t_b ON t (b)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_up', NULL, 'UPPER(c) + DTOS(d)', NULL, 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_fd', NULL, 'c', '.NOT. DELETED()', 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_fn', NULL, 'n', 'n > 0', 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_u', NULL, 'c', NULL, 1, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_dc', NULL, 'c', NULL, 8, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('t', 't_st', NULL, 'STR(n) + c', 'l .OR. DELETED()', 0, 0, NULL)
sql CREATE TABLE w (id INTEGER, c CHAR(10), n NUMERIC(8, 2), d DATE, l LOGICAL, b DOUBLE(2), m MEMO, bl BLOB)
sql CREATE INDEX w_id ON w (id)
sql CREATE INDEX w_c ON w (c)
sql CREATE INDEX w_n ON w (n)
sql CREATE INDEX w_d ON w (d)
sql CREATE INDEX w_l ON w (l)
sql CREATE INDEX w_b ON w (b)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_up', NULL, 'UPPER(c) + DTOS(d)', NULL, 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_fd', NULL, 'c', '.NOT. DELETED()', 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_fn', NULL, 'n', 'n > 0', 0, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_u', NULL, 'c', NULL, 1, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_dc', NULL, 'c', NULL, 8, 0, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('w', 'w_st', NULL, 'STR(n) + c', 'l .OR. DELETED()', 0, 0, NULL)
sql CREATE TABLE big (id INTEGER, c CHAR(10), n NUMERIC(8, 2))
sql CREATE INDEX big_id ON big (id)
sql CREATE INDEX big_c ON big (c)
sql CREATE INDEX big_n ON big (n)
