# An ADT table of every field type: a record with a value in every field,
# one of NULLs, one of empty values, one of boundary values, one deleted.

# Double and CurDouble fields of each count of decimals.
createtable dd.adt 0 d0,Double,0;d1,Double,1;d5,Double,5;d15,Double,15;d16,Double,16;d17,Double,17;d20,Double,20;c0,CurDouble,0;c4,CurDouble,4

# Names longer than ten characters, two of them the same in their first ten.
createtable nm.adt 0 a_long_field_name,Char,2;a_long_field_other,Char,2;Mixed,Char,2
append
set #1 string a
set #2 string b
set #3 string c
write
close
# Values a converted field cannot hold: a Double wider than its Numeric;
# a CHAR wider than a DBF Character.
createtable big.adt 0 k,Char,2;dbl,Double,2
append
set k string a
set dbl double 1.5
write
append
set k string b
set dbl double 1e300
write
append
set k string c
set dbl double 2.5
write
close
createtable wide.adt 0 k,Char,2;w,Char,65000
append
set k string a
set w string \@00300w
write
close
# AutoInc and RowVersion values that are not the records' numbers: the
