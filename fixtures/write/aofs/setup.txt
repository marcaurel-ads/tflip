# AOFs: a table with orders of many kinds, and the same in FoxPro, Visual
# FoxPro and Clipper tables.
sql CREATE TABLE p (id INTEGER, name CHAR(10), ciname CICHAR(10), amount MONEY, born DATE, active LOGICAL, code INTEGER, k2 INTEGER)
sql CREATE INDEX id ON p (id)
sql CREATE INDEX name ON p (name)
sql CREATE INDEX ciname ON p (ciname)
sql CREATE INDEX born ON p (born)
sql CREATE INDEX active ON p (active)
sql CREATE INDEX amt ON p (amount DESC)
sql CREATE INDEX name_id ON p (name, id)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('p', 'p', 'upname', 'UPPER(name)', NULL, 2, 512, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('p', 'p', 'condk2', 'k2', 'k2 > 3', 2, 512, NULL)
sql EXECUTE PROCEDURE sp_CreateIndex90 ('p', 'p', 'custcode', 'code', NULL, 6, 512, NULL)
