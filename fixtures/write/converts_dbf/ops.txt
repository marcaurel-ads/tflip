# Each DBF table converted to every table type, its own included.
tabletype ntx
open n.dbf
convert 2 ntx n_n.dbf
convert 2 cdx n_c.dbf
convert 2 vfp n_v.dbf
convert 2 adt n_a.adt
close
tabletype cdx
open c.dbf
convert 2 ntx c_n.dbf
convert 2 cdx c_c.dbf
convert 2 vfp c_v.dbf
convert 2 adt c_a.adt
close
tabletype vfp
open v.dbf
convert 2 ntx v_n.dbf
convert 2 cdx v_c.dbf
convert 2 vfp v_v.dbf
convert 2 adt v_a.adt
close
open vn.dbf
convert 2 ntx vn_n.dbf
convert 2 cdx vn_c.dbf

close

# A Double too wide for the Numeric a FoxPro table makes of it (5179).

close
# A file that is there already is overwritten.
tabletype vfp
open v.dbf
convert 2 ntx old.dbf
convert 2 vfp v_v.dbf
close
