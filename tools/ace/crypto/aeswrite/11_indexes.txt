sql EXECUTE PROCEDURE sp_ModifyDatabase('ENCRYPT_INDEXES', 'TRUE')
sql CREATE TABLE ti (id INTEGER, name CHAR(10))
sql CREATE INDEX ti_name ON ti (name)
sql INSERT INTO ti VALUES (1, 'one')
