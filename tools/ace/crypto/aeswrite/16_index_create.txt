sql EXECUTE PROCEDURE sp_ModifyDatabase('ENCRYPT_INDEXES', 'TRUE')
sql EXECUTE PROCEDURE sp_EncryptTable('ti')
sql INSERT INTO ti VALUES (4, 'four')

sql INSERT INTO ti VALUES (5, 'five')
