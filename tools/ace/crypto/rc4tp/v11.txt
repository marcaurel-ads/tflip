connect Data Source=%D\v11\free;ServerType=LOCAL;
createt 4 %D\v11\free\fv
createt 2 %D\v11\free\fc
stmttype 4
sql INSERT INTO fv VALUES (1, 'free memo')
sql INSERT INTO fv VALUES (2, 'second')
stmttype 2
sql INSERT INTO fc VALUES (1, 'free memo')
disconnect
connect Data Source=%D\v11\;ServerType=LOCAL;EncryptionType=AES128;FIPS=FALSE;DDPassword=ddpw;
sql CREATE DATABASE "d.add" ENCRYPTIONTYPE AES128 DDPASSWORD 'ddpw'
disconnect
connect Data Source=%D\v11\d.add;ServerType=LOCAL;User ID=ADSSYS;EncryptionType=AES128;FIPS=FALSE;DDPassword=ddpw;

tget tc 214
