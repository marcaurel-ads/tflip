# Maintenance by an exclusive handle without the password on an AES table, and what a handle with it finds after.
include setup.inc
open h2 c2 %D\t.adt 1
order y h2 ICH
order yn h2 IN1
tablecheck h2
reindex h2
tablecheck h2
top y
recno y
keycount y
index inew h2 %D\t.adi INEW 0 N1
index inew2 h2 %D\tb.adi INEW2 0 CH
dropindex yn
top h2
dump h2
pack h2
count h2
close h2
open h1 c1 %D\t.adt 4
enable h1 secret
tablecheck h1
count h1
order x1 h1 ICH
order x1n h1 IN1
order xnew h1 INEW
order xnew2 h1 INEW2
top x1
recno x1
extract x1
keycount x1
seek x1 bravo 0
dump h1
goto h1 2
dump h1
goto h1 3
dump h1
