# Query logging with an encryption password on a dictionary connection: an AES-128 dictionary whose tables are in the clear (j) and a default one (r); the log table, and enabling it again with no password and a wrong one.
connect c0 Data Source=%D;ServerType=LOCAL;EncryptionType=AES128;FIPS=FALSE;DDPassword=ddpw;
sql c0 CREATE DATABASE "j.add" ENCRYPTIONTYPE AES128 DDPASSWORD 'ddpw'
disconnect c0
connect c1 Data Source=%D\j.add;ServerType=LOCAL;User ID=ADSSYS;EncryptionType=AES128;FIPS=FALSE;DDPassword=ddpw;
sql c1 CREATE TABLE a (id INTEGER)
sql c1 EXECUTE PROCEDURE sp_EnableQueryLogging('ql', 0, 0, 0, 'pw')
sql c1 INSERT INTO a VALUES (1)
sql c1 SELECT * FROM a
sqlc c1 EXECUTE PROCEDURE sp_GetQueryLoggingResults('pw')
sqlc c1 EXECUTE PROCEDURE sp_GetQueryLoggingResults('wrong')
sqlc c1 EXECUTE PROCEDURE sp_GetQueryLoggingResults('')
sql c1 EXECUTE PROCEDURE sp_DisableQueryLogging()
openn q c1 ql 4
tablecheck q
count q
close q
sql c1 EXECUTE PROCEDURE sp_EnableQueryLogging('ql', 0, 0, 0, '')
sql c1 SELECT * FROM a
sql c1 EXECUTE PROCEDURE sp_DisableQueryLogging()
sql c1 EXECUTE PROCEDURE sp_EnableQueryLogging('ql', 0, 0, 0, 'wrong')
sql c1 SELECT * FROM a
sql c1 EXECUTE PROCEDURE sp_DisableQueryLogging()
sql c1 EXECUTE PROCEDURE sp_EnableQueryLogging('ql', 0, 0, 0, 'pw')
sql c1 SELECT * FROM a
sql c1 EXECUTE PROCEDURE sp_DisableQueryLogging()
sql c1 EXECUTE PROCEDURE sp_EnableQueryLogging('ql', 1, 0, 0, '')
sql c1 SELECT * FROM a
sql c1 EXECUTE PROCEDURE sp_DisableQueryLogging()
openn q c1 ql 4
tablecheck q
count q
close q
disconnect c1
