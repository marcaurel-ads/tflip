# An RC4 free table opened, written and dropped through a statement given its password (AdsStmtSetTablePassword), and what AdsStmtEnableEncryption does.
connect Data Source=%D;ServerType=LOCAL;
sql CREATE TABLE r (id INTEGER, name CHAR(10))
sql INSERT INTO r VALUES (1, 'one')
sql CREATE TABLE p (id INTEGER, name CHAR(10))
sql INSERT INTO p VALUES (1, 'plain')
topen %D\r.adt
t.enable pw
tencrypt
tclose
# Without the password, and with a wrong one.
sql SELECT * FROM r
sql INSERT INTO r VALUES (9, 'nine')
stmt
s.exec SELECT * FROM r
s.setpw r wrong
s.exec SELECT * FROM r
# With it: reads, writes and DROP; another statement has none.
s.setpw r pw
s.exec SELECT * FROM r
s.exec INSERT INTO r VALUES (2, 'two')
s.exec UPDATE r SET name = 'x' WHERE id = 1
s.exec SELECT * FROM r
s.exec SELECT * FROM r r1, r r2
s.exec CREATE TABLE r9 (id INTEGER, name CHAR(10))
s.exec INSERT INTO r9 SELECT * FROM r
s.exec SELECT * FROM r9
sql SELECT * FROM r
s.clear
s.exec SELECT * FROM r

# A table that is not encrypted opens with any password; a join needs each one's.
