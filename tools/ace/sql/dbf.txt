# Blank values: a Clipper or FoxPro table has no NULL, so an empty field
# (blank, 0, false, spaces) is NULL to IS NULL, IFNULL and COALESCE, and its
# value (a blank one's 0, empty date, false or spaces) to comparisons,
# arithmetic and aggregates; a blank one shows as NULL (but text, doubles and
# integers) and is a group of its own. A VFP table's blank values are values
# (a date 1899-12-30), but IFNULL takes an empty one for NULL.
tabletype 1
direct CREATE TABLE ntx_b (id integer, c char(6), v varchar(8), d date, n numeric(6,2), l logical, f double, g bigint, i integer)
directnull INSERT INTO ntx_b (id, c, v, d, n, l, f, g, i) VALUES (1, 'x', 'x', '2020-01-02', 5, true, 1.5, 7, 3)
directnull INSERT INTO ntx_b (id) VALUES (2)
directnull INSERT INTO ntx_b (id, c, v, d, n, l, f, g, i) VALUES (3, 'y', 'y', '2021-01-01', 0, false, 2.5, 9, 4)
query SELECT COUNT(*), COUNT(c), COUNT(v), COUNT(d), COUNT(n), COUNT(l), COUNT(f), COUNT(g), COUNT(i) FROM ntx_b
query SELECT MIN(f), MAX(f), MIN(g), MAX(g), MIN(i), MAX(i), MIN(c), MAX(c), MIN(d), MAX(d), MIN(l), MAX(l) FROM ntx_b
query SELECT SUM(f), SUM(g), SUM(i), AVG(f), AVG(g), AVG(i), SUM(n), AVG(n), MIN(n), MAX(n) FROM ntx_b
query SELECT MIN(n), MAX(n), SUM(n), AVG(n), COUNT(n), MIN(d), MAX(d), MIN(f), SUM(i), MIN(c), COUNT(c), MIN(l), MAX(l), MAX(i) FROM ntx_b WHERE id = 2
query SELECT COUNT(n + 0), MIN(n + 0), SUM(n * 2), COUNT(UPPER(v)) FROM ntx_b
query SELECT id, IFNULL(n, 9), IFNULL(c, 'z'), IFNULL(l, TRUE), COALESCE(n, 7), COALESCE(l, TRUE) FROM ntx_b
query SELECT n + 0, d, c, n, l, f, CASE WHEN n IS NULL THEN 'null' ELSE 'not' END FROM ntx_b WHERE id = 2
query SELECT {static} id, n, d, c, v, l, f, g, i FROM ntx_b WHERE id = 2
query SELECT id FROM ntx_b WHERE n < 1
query SELECT id FROM ntx_b WHERE n = 0
query SELECT id FROM ntx_b WHERE d < '2000-01-01'
query SELECT id FROM ntx_b WHERE d IS NULL
query SELECT id FROM ntx_b WHERE c = ''
query SELECT id FROM ntx_b WHERE l = FALSE
query SELECT id FROM ntx_b WHERE f = 0 AND i = 0
query SELECT COUNT(*) FROM ntx_b WHERE n IS NOT NULL
query SELECT COUNT(*) FROM ntx_b WHERE n + 0 IS NULL
query SELECT COUNT(*) FROM ntx_b WHERE n IN (0)
query SELECT n, COUNT(*), COUNT(n) FROM ntx_b GROUP BY n
query SELECT n + 0, COUNT(*) FROM ntx_b GROUP BY n + 0
query SELECT d, COUNT(*), COUNT(d) FROM ntx_b GROUP BY d
query SELECT c, COUNT(*), COUNT(c) FROM ntx_b GROUP BY c
query SELECT l, COUNT(*), COUNT(l) FROM ntx_b GROUP BY l
query SELECT id, COUNT(n), SUM(i) FROM ntx_b GROUP BY id
query SELECT DISTINCT n FROM ntx_b
query SELECT COUNT(a.n), MIN(b.d) FROM ntx_b a, ntx_b b WHERE a.id = b.id
query SELECT id, n FROM ntx_b UNION SELECT 9, 0 FROM system.iota
query SELECT MAX(n) - MIN(n), COUNT(*) - COUNT(n) FROM ntx_b
query SELECT MIN(x.n), COUNT(x.n) FROM (SELECT n FROM ntx_b) x
