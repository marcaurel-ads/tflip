# Rights on functions and packages, and to create them, held through a group:
# what a member of the group may alter, rename, drop and create. The
# dictionary, fg/fg.add, is kept in fixtures/sql/ddfuncgroup/:
#
#     run.sh <work> <ace32.dll> ddfuncgroup.txt > ../../../fixtures/sql/ddfuncgroup.json
#     cp <work>/data/fg/* ../../../fixtures/sql/ddfuncgroup/
ddcreate fg\fg.add
stmt
direct CREATE FUNCTION fa(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION fb(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION fc(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE PACKAGE pa
direct CREATE PACKAGE pb
direct CREATE FUNCTION pa.x(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct EXECUTE PROCEDURE sp_CreateUser('u1', 'pw1', '')
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g1', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g2', '')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u1', 'g1')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u2', 'g2')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u2', 'g1')
direct GRANT ALTER ON fa TO g1
direct GRANT DROP ON fb TO g1
direct GRANT ALTER ON pa TO g1
direct GRANT DROP ON pb TO g1
direct GRANT CREATE FUNCTION TO g2
direct GRANT CREATE PACKAGE TO g2
direct GRANT EXECUTE ON fc TO g1 WITH GRANT
disconnect
connect fg\fg.add u1 pw1
stmt
direct ALTER FUNCTION fa(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct ALTER FUNCTION fa fa1(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct ALTER FUNCTION fb(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct ALTER FUNCTION pa.x(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct ALTER FUNCTION pa.x pa.x1(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct CREATE FUNCTION pa.y(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct CREATE FUNCTION pb.y(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct CREATE FUNCTION fu1(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct CREATE PACKAGE pu1
direct DROP FUNCTION fb
direct DROP FUNCTION fa1
direct DROP PACKAGE pb
direct DROP PACKAGE pa
direct GRANT EXECUTE ON fc TO u3
direct GRANT EXECUTE ON fc TO u3 WITH GRANT
ddrename fb fb2 18 0
ddrename pa pa2 19 0
disconnect
connect fg\fg.add u2 pw2
stmt
direct CREATE FUNCTION fu1(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct CREATE PACKAGE pu1
direct CREATE FUNCTION pu1.z(a INTEGER) RETURNS INTEGER BEGIN RETURN 2; END
direct ALTER FUNCTION fu1(a INTEGER) RETURNS INTEGER BEGIN RETURN 3; END
direct ALTER FUNCTION fc(a INTEGER) RETURNS INTEGER BEGIN RETURN 3; END
direct DROP FUNCTION fu1
direct DROP FUNCTION fc
direct DROP PACKAGE pu1
disconnect
connect fg\fg.add adssys
stmt
query SELECT Name, Package FROM system.functions
query SELECT Name FROM system.packages
query SELECT * FROM system.permissions
disconnect
