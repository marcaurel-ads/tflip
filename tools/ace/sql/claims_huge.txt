## group huge
direct CREATE TABLE sq (n INTEGER)
direct DECLARE i INTEGER; SET i = 0; WHILE i <= 127 DO INSERT INTO sq VALUES (i); SET i = i + 1; END WHILE;
direct CREATE TABLE s3 (n INTEGER)
direct DECLARE i INTEGER; SET i = 1; WHILE i <= 3000 DO INSERT INTO s3 VALUES (i); SET i = i + 1; END WHILE;
direct CREATE TABLE huge (id INTEGER, grp INTEGER, city CHAR(10), d DATE)
direct INSERT INTO huge SELECT a.n * 3000 + b.n, MOD(a.n * 3000 + b.n, 10), 'City' + CAST(MOD(a.n * 3000 + b.n, 50) AS SQL_VARCHAR(2)), CAST('2019-01-01' AS SQL_DATE) + MOD(a.n * 3000 + b.n, 1000) FROM sq a, s3 b
direct CREATE INDEX HID ON huge (id)
direct CREATE INDEX HGRP ON huge (grp)
direct CREATE INDEX HCITY ON huge (city)
direct CREATE INDEX HD ON huge (d)
