# Which views open a live cursor, and what a client can do with one: a
# dictionary made here (vw.add, next to the free tables) with views over
# indexed.adt (it has indexes) and cust.adt. peek is AdsSqlPeekStatement's
# live flag. A view takes no ORDER BY without TOP (AdsDDAddView fails with
# 2117, as the end shows), so no view here sorts.
#
#     run.sh <work> <ace32.dll> views.txt > ../../../fixtures/sql/views.json
ddcreate vw.add
stmt
ddview v_all SELECT * FROM indexed
ddview v_sub SELECT id, name FROM indexed
ddview v_where SELECT id, name FROM indexed WHERE id > 1
ddview v_alias SELECT id AS ident, name AS label FROM indexed
ddview v_of_view SELECT id, name FROM v_sub
ddview v_of_view_where SELECT id FROM v_where WHERE id < 4
ddview v_expr SELECT id, UPPER(name) AS n FROM indexed
ddview v_top SELECT TOP 2 id, name FROM indexed
ddview v_topord SELECT TOP 2 id, name FROM indexed ORDER BY name
ddview v_dist SELECT DISTINCT name FROM indexed
ddview v_join SELECT i.id, c.name FROM indexed i JOIN cust c ON c.id = i.id
ddview v_like SELECT id, name FROM indexed WHERE name LIKE 'A%'
ddview v_static SELECT {static} id, name FROM indexed
ddview v_cust SELECT * FROM cust
ddview v_dup SELECT id, id AS id2 FROM indexed
ddview v_union SELECT id FROM indexed UNION SELECT id FROM cust
ddview v_from SELECT * FROM (SELECT id, name FROM indexed) t
ddview v_grp SELECT name, COUNT(*) AS n FROM indexed GROUP BY name
# The plans.
query SHOW PLAN FOR SELECT * FROM v_sub

query SHOW PLAN FOR SELECT * FROM v_where WHERE id < 4 ORDER BY name
query SHOW PLAN FOR SELECT * FROM v_of_view_where
query SHOW PLAN FOR SELECT * FROM v_expr

# A view's ORDER BY: AdsDDAddView and CREATE VIEW refuse one without TOP
# (2117), whatever it sorts by.
ddview v_ord SELECT id, name FROM indexed ORDER BY name
ddview v_ord SELECT id, name FROM indexed ORDER BY id DESC
ddview v_ord SELECT id, name FROM indexed WHERE id > 1 ORDER BY 1
ddview v_ord SELECT * FROM v_sub ORDER BY name
ddview v_ord SELECT id FROM indexed UNION SELECT id FROM cust ORDER BY 1
ddview v_ord4 SELECT * FROM (SELECT id FROM indexed ORDER BY id) t
ddview v_ord SELECT DISTINCT name FROM indexed ORDER BY name
ddview v_ord SELECT TOP 2 id FROM indexed ORDER BY id DESC
ddview v_ord3 SELECT TOP 100 PERCENT id FROM indexed ORDER BY id
ddview v_ord2 SELECT id FROM indexed ORDER BY id
ddview v_ord2 SELECT id FROM indexed
