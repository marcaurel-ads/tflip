# sp_DisableTriggers and sp_EnableTriggers by a user who is not the
# administrator and holds a right on the table: which rights let it, for a
# table, a trigger and all users' triggers. The dictionary, gq/gq.add, is kept
# in fixtures/sql/ddtrigrights/:
#
#     run.sh <work> <ace32.dll> ddtrigrights.txt > ../../../fixtures/sql/ddtrigrights.json
#     cp <work>/data/gq/* ../../../fixtures/sql/ddtrigrights/
ddcreate gq\gq.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10)) IN DATABASE
direct CREATE TABLE audit (a INTEGER) IN DATABASE
direct CREATE TRIGGER tr1 ON t1 AFTER INSERT BEGIN INSERT INTO audit VALUES (1); END
direct EXECUTE PROCEDURE sp_CreateUser('u1', 'pw1', '')
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateUser('u4', 'pw4', '')
direct GRANT SELECT, INSERT, UPDATE, DELETE ON t1 TO u1
direct GRANT SELECT, ALTER ON t1 TO u2
direct GRANT SELECT, ALTER, DROP ON t1 TO u3
direct GRANT SELECT, INSERT ON audit TO u1
direct GRANT SELECT, INSERT ON audit TO u2
direct GRANT SELECT, INSERT ON audit TO u3
direct GRANT SELECT, INSERT ON audit TO u4
direct GRANT SELECT, INSERT, UPDATE, DELETE ON t1 TO u4
direct GRANT CREATE TABLE TO u4
direct GRANT CREATE PROCEDURE TO u4
direct EXECUTE PROCEDURE sp_ModifyDatabase('VERIFY_ACCESS_RIGHTS', 'T')
disconnect
#
connect gq\gq.add u1 pw1
stmt
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('tr1', 't1', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('', '', 1, 0)
disconnect
connect gq\gq.add u2 pw2
stmt
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 0, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('t1', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 1, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('t1', '', 1, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('tr1', 't1', 0, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('tr1', 't1', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('tr1', 't1', 1, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('tr1', 't1', 1, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('', '', 1, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('', '', 0, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('', '', 1, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('audit', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('nope', '', 0, 0)
query SELECT Name, Triggers_Disabled FROM system.tables
disconnect
connect gq\gq.add u3 pw3
stmt
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 1, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('t1', '', 1, 0)
direct EXECUTE PROCEDURE sp_EnableTriggers('t1', '', 0, 0)
disconnect
connect gq\gq.add u4 pw4
stmt
direct EXECUTE PROCEDURE sp_DisableTriggers('t1', '', 0, 0)
direct EXECUTE PROCEDURE sp_DisableTriggers('', '', 1, 0)
disconnect
connect gq\gq.add adssys
stmt
query SELECT Name, Triggers_Disabled FROM system.tables
disconnect
