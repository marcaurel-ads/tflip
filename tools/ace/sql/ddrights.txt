# What a user who is not the administrator may do to a dictionary's tables,
# indexes, views, procedures and triggers (and users, groups, links and
# integrity rules by the system procedures), before and after the rights
# GRANT gives: CREATE <type> and the rights on an object (ALTER, DROP), to
# the user, to a group and to DB:Public. The dictionary, dr/dr.add,
# is kept in fixtures/sql/ddrights/:
#
#     run.sh <work> <ace32.dll> ddrights.txt > ../../../fixtures/sql/ddrights.json
#     cp <work>/data/dr/* ../../../fixtures/sql/ddrights/
ddcreate dr\dr.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10), PRIMARY KEY (id))
direct CREATE TABLE t2 (id INTEGER, pid INTEGER)
direct CREATE TABLE t3 (id INTEGER)
direct EXECUTE PROCEDURE sp_CreateLink('lk0', 'dr.add', 1, 0, 0, 'ADSSYS', '')
direct CREATE INDEX t2pid ON t2 (pid)
direct CREATE VIEW vw1 AS SELECT id FROM t1

direct CREATE VIEW vw3 AS SELECT id FROM t1

direct CREATE PROCEDURE pr2() BEGIN SELECT 1 FROM system.iota; END
direct CREATE PROCEDURE pr3() BEGIN SELECT 1 FROM system.iota; END
direct CREATE TRIGGER tg1 ON t1 AFTER INSERT BEGIN SELECT 1 FROM system.iota; END
direct CREATE TRIGGER tg2 ON t1 AFTER UPDATE BEGIN SELECT 1 FROM system.iota; END
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateUser('u4', 'pw4', '')
direct EXECUTE PROCEDURE sp_CreateUser('u5', 'pw5', '')

direct EXECUTE PROCEDURE sp_CreateGroup('g1', '')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u3', 'g1')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u6', 'g1')

direct GRANT SELECT ON t2 TO u2
direct GRANT SELECT ON vw1 TO u2
