# The scalar functions DATABASE(), NEWID(), NEWIDSTRING() and RAND() on a
# free-table connection, for fixtures/sql/funcs.json (an empty data folder;
# they read no table but RAND's own). DATABASE() is the connection folder
# (written <data>); NEWID() is a GUID, NEWIDSTRING() its text -- both random,
# normalized on replay.
stmt
query SELECT DATABASE() FROM system.iota
query SELECT NEWID() FROM system.iota
query SELECT NEWIDSTRING() FROM system.iota
# NEWID and NEWIDSTRING take no argument: the engine has no format variants
# and rejects one with 2159 (not documented in the CHM; probed on the VM).
query SELECT NEWIDSTRING(0) FROM system.iota
query SELECT NEWID(0) FROM system.iota
# RAND([seed]) is a DOUBLE in (0, 1) from the connection's own generator: a
# seed restarts it, RAND() goes on from where it is, across the connection's
# statements; a new connection's starts in the same state. (RAND(0) seeds
# from the clock, so it is not recorded.)
query SELECT RAND(), RAND(), RAND() FROM system.iota
