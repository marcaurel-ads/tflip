# Stored procedures that read tables and their input parameters through
# __input, and the system variables in them, in a dictionary of an empty
# folder, pv/pv.add:
#
#     run.sh <work> <ace32.dll> procvars.txt > ../../../fixtures/sql/procvars.json
ddcreate pv\pv.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10))
direct INSERT INTO t1 VALUES (1, 'one')
direct INSERT INTO t1 VALUES (2, 'two')
direct INSERT INTO t1 VALUES (3, 'three')
#
# __input: a table of one row, the input parameters as its columns.
direct CREATE PROCEDURE i1(a INTEGER, b CHAR(10), c DATE, o1 INTEGER OUTPUT, o2 CHAR(10) OUTPUT) BEGIN INSERT INTO __output SELECT a, b FROM __input; END
query EXECUTE PROCEDURE i1(5, 'hello', '2020-01-02')
direct CREATE PROCEDURE i3(a INTEGER, o INTEGER OUTPUT) BEGIN INSERT INTO __output SELECT COUNT(*) FROM __input; END
query EXECUTE PROCEDURE i3(1)

query EXECUTE PROCEDURE i4(5, 'x')
query EXECUTE PROCEDURE i4(2, 'x')

#
# The statements a procedure runs, and what it reads of its own and its
# connection.
direct CREATE PROCEDURE v1(r CHAR(80) OUTPUT) BEGIN DECLARE u INTEGER; INSERT INTO t1 VALUES (31, 'x'); INSERT INTO t1 VALUES (32, 'y'); u = ::stmt.UpdateCount; INSERT INTO __output SELECT 'uc ' + TRIM(CAST(u AS SQL_CHAR)) FROM system.iota; END
query EXECUTE PROCEDURE v1()
direct CREATE PROCEDURE v2(r CHAR(80) OUTPUT) BEGIN DECLARE u INTEGER; UPDATE t1 SET name = 'q' WHERE id > 30; DELETE FROM t1 WHERE id = 32; u = ::stmt.UpdateCount; INSERT INTO __output SELECT 'uc ' + TRIM(CAST(u AS SQL_CHAR)) FROM system.iota; END
query EXECUTE PROCEDURE v2()
direct CREATE PROCEDURE v3(r CHAR(80) OUTPUT) BEGIN INSERT INTO __output SELECT IFNULL(::stmt.TrigName, 'null') + '|' + TRIM(CAST(::stmt.TrigRecNo AS SQL_CHAR)) + '|' + TRIM(CAST(::conn.TransactionCount AS SQL_CHAR)) + '|' + ::conn.Collation FROM system.iota; END
query EXECUTE PROCEDURE v3()
direct CREATE PROCEDURE v4(r CHAR(80) OUTPUT) BEGIN INSERT INTO __output SELECT LEFT(::stmt.Name, 1) + '|' + LEFT(::conn.Name, 4) + '|' + TRIM(CAST(LENGTH(::stmt.Name) AS SQL_CHAR)) FROM system.iota; END
query EXECUTE PROCEDURE v4()
direct CREATE PROCEDURE v6(r CHAR(80) OUTPUT) BEGIN INSERT INTO t1 VALUES (41, 'a'); INSERT INTO __output SELECT TRIM(CAST(::stmt.UpdateCount AS SQL_CHAR)) FROM system.iota; END

#
# Output parameters of every type, and the input parameters that are __input's columns
# (a procedure's parameters are kept in 275 bytes of the dictionary).
direct CREATE PROCEDURE ot0(a INTEGER, o1 INTEGER OUTPUT, o2 SHORT OUTPUT, o3 LONGINT OUTPUT, o4 DOUBLE OUTPUT, o5 NUMERIC(8,2) OUTPUT, o6 MONEY OUTPUT) BEGIN INSERT INTO __output VALUES (1, 2, 3, 4.5, 6.25, 7.5); END
