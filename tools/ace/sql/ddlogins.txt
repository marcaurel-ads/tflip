# Disabled logins: what a connection to a dictionary answers while the
# database's logins are disabled (sp_ModifyDatabase 'LOGINS_DISABLED') for the
# administrator and for another user, with and without a message, and for a
# user whose own logins are disabled (sp_ModifyUserProperty). The dictionary,
# lg/lg.add, is kept in fixtures/sql/ddlogins/:
#
#     run.sh <work> <ace32.dll> ddlogins.txt > ../../../fixtures/sql/ddlogins.json
#     cp <work>/data/lg/* ../../../fixtures/sql/ddlogins/
ddcreate lg\lg.add
stmt
direct CREATE TABLE t1 (id INTEGER) IN DATABASE
direct INSERT INTO t1 VALUES (1)
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct GRANT SELECT ON t1 TO u2
direct GRANT SELECT ON t1 TO u3
direct EXECUTE PROCEDURE sp_ModifyDatabase('LOGINS_DISABLED', 'T')
query SELECT Logins_Disabled, Logins_Disabled_Msg FROM system.dictionary
disconnect
#
# A statement made on a connection before the logins were disabled is not
# tested here; a new connection is.
connect lg\lg.add adssys
stmt
query SELECT Logins_Disabled, Logins_Disabled_Msg FROM system.dictionary
query SELECT * FROM t1
disconnect
connect lg\lg.add u2 pw2
stmt
disconnect
connect lg\lg.add u3 pw3
stmt
disconnect
connect lg\lg.add u2 wrong
stmt
disconnect
connect lg\lg.add nosuch pw
stmt
disconnect
#
# The message.
connect lg\lg.add adssys
stmt
direct EXECUTE PROCEDURE sp_ModifyDatabase('LOGINS_DISABLED_MSG', 'back at noon')
query SELECT Logins_Disabled, Logins_Disabled_Msg FROM system.dictionary
disconnect
connect lg\lg.add u2 pw2
stmt
disconnect
connect lg\lg.add u3 pw3
stmt
disconnect
connect lg\lg.add u2 wrong
stmt
disconnect
#
# Enabled again.
connect lg\lg.add adssys
stmt
direct EXECUTE PROCEDURE sp_ModifyDatabase('LOGINS_DISABLED', 'F')
query SELECT Logins_Disabled, Logins_Disabled_Msg FROM system.dictionary
disconnect
connect lg\lg.add u2 pw2
stmt
query SELECT * FROM t1
disconnect
#
# One user's logins disabled.
connect lg\lg.add adssys
stmt
direct EXECUTE PROCEDURE sp_ModifyUserProperty('u2', 'LOGINS_DISABLED', 'T')
query SELECT Name, Logins_Disabled FROM system.users
disconnect
connect lg\lg.add u2 pw2
stmt
disconnect
connect lg\lg.add u2 wrong
stmt
disconnect
connect lg\lg.add u3 pw3
stmt
query SELECT * FROM t1
disconnect
connect lg\lg.add adssys
stmt
direct EXECUTE PROCEDURE sp_ModifyUserProperty('u2', 'LOGINS_DISABLED', 'F')
disconnect
connect lg\lg.add u2 pw2
stmt
query SELECT * FROM t1
disconnect
