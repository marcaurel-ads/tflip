# Table triggers' CONFLICT type and NO MEMOS / NO TRANSACTION options, on a
# copy of the psm dictionary (fixtures/adt/psm) alone, whose own triggers it
# drops first. Recorded on the remote server (the evaluation server under
# Wine), whose transactions tflip has:
#
#     SQL_SERVER_TYPE=2 run.sh <work> <ace32.dll> triggers.txt <share>
#             > ../../../fixtures/sql/triggers.json
#
# (<share> the server's path to the work folder's data/, e.g.
# \\127.0.0.1:6262\data\trg\data.) On the local server the same script
# differs only in the transactions: see docs/TRANSACTIONS.md. The psm
# dictionary has no table with a BLOB; probed apart, a BLOB reads as NULL
# under NO MEMOS as a memo does.
# The closed cursors' tables are not kept open: a table kept open keeps the
# triggers it had when it was opened (see docs/ADS_BUGS.md).
cachecursors 0
connect psm.add ADSSYS adssyspw
stmt
direct DROP TRIGGER items.items_bi
direct DROP TRIGGER items.items_ai
direct DROP TRIGGER items.items_au
direct DROP TRIGGER items.items_iod
direct DROP TRIGGER items.items_conflict
direct DELETE FROM items
direct DELETE FROM audit
#
# CONFLICT triggers: UPDATE and DELETE only, one of each a table; stored
# with the trigger type 8; never fired by the table's own changes.
direct CREATE TRIGGER c_ins ON items CONFLICT INSERT BEGIN INSERT INTO audit (msg) VALUES ('ci'); END
direct CREATE TRIGGER c_upd ON items CONFLICT UPDATE BEGIN INSERT INTO audit (msg) VALUES ('cu'); END
direct CREATE TRIGGER c_del ON items CONFLICT DELETE BEGIN INSERT INTO audit (msg) VALUES ('cd'); END NO MEMOS NO TRANSACTION PRIORITY 3
direct CREATE TRIGGER c_upd2 ON items CONFLICT UPDATE BEGIN INSERT INTO audit (msg) VALUES ('cu2'); END
direct CREATE TRIGGER c_bad ON items CONFLICT INSTEAD OF UPDATE BEGIN END
query SELECT name, trig_event_type, trig_trigger_type, trig_container_type, trig_options, trig_priority FROM system.triggers WHERE name = 'c_upd'
query SELECT name, trig_event_type, trig_trigger_type, trig_container_type, trig_options, trig_priority FROM system.triggers WHERE name = 'c_del'
direct INSERT INTO items VALUES (1, 'a', 0, 'memo one')
direct UPDATE items SET name = 'b'
direct DELETE FROM items
query SELECT msg FROM audit
direct DROP TRIGGER items.c_upd
direct DROP TRIGGER items.c_del
