# What the grant option holds a user who is not the administrator to on
# functions, packages, links, users and groups: the texts of the refusals,
# which of them name the object, and what is cut from ALL. The dictionary,
# g2/g2.add, is kept in fixtures/sql/ddgrantopt2/:
#
#     run.sh <work> <ace32.dll> ddgrantopt2.txt > ../../../fixtures/sql/ddgrantopt2.json
#     cp <work>/data/g2/* ../../../fixtures/sql/ddgrantopt2/
ddcreate g2\g2.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10))
direct CREATE FUNCTION f1(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION f2(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE PACKAGE pk1
direct CREATE PACKAGE pk2
direct CREATE FUNCTION pk1.pf(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE PROCEDURE pr1() BEGIN SELECT 1 FROM system.iota; END
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateUser('u5', 'pw5', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g1', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g2', '')
direct EXECUTE PROCEDURE sp_CreateLink('lk1', 'g2.add', 1, 0, 0, 'ADSSYS', '')
direct EXECUTE PROCEDURE sp_CreateLink('lk2', 'g2.add', 1, 0, 0, 'ADSSYS', '')
direct GRANT EXECUTE ON f1 TO u2 WITH GRANT
direct GRANT EXECUTE ON f2 TO u2
direct GRANT ALTER ON f2 TO u2 WITH GRANT
direct GRANT EXECUTE ON pk1 TO u2 WITH GRANT
direct GRANT EXECUTE ON pk2 TO u2
direct GRANT ALTER ON lk1 TO u2 WITH GRANT
direct GRANT ALTER ON lk2 TO u2
direct GRANT ALTER ON u3 TO u2 WITH GRANT
direct GRANT ALTER ON u5 TO u2
direct GRANT ALTER ON g1 TO u2 WITH GRANT
direct GRANT ALTER ON g2 TO u2
query SELECT * FROM system.permissions
disconnect

direct GRANT ALL ON lk2 TO u5
direct GRANT ALTER ON u3 TO u5
direct GRANT ALTER ON u5 TO u3

direct GRANT ALL ON u3 TO g2

direct GRANT ALTER ON g2 TO u5
