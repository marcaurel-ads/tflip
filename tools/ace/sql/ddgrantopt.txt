# What the grant option lets a user who is not the administrator GRANT and
# REVOKE: the rights it holds with the grant option (its own, a column's, a
# group's, to create objects), and nothing else. The dictionary, go/go.add,
# is kept in fixtures/sql/ddgrantopt/:
#
#     run.sh <work> <ace32.dll> ddgrantopt.txt > ../../../fixtures/sql/ddgrantopt.json
#     cp <work>/data/go/* ../../../fixtures/sql/ddgrantopt/
ddcreate go\go.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10))
direct CREATE TABLE t2 (id INTEGER)
direct CREATE TABLE t3 (id INTEGER)
direct CREATE TABLE t4 (id INTEGER, name CHAR(10))
direct CREATE VIEW vw1 AS SELECT id FROM t1
direct CREATE VIEW vw2 AS SELECT id FROM t1
direct CREATE PROCEDURE pr1() BEGIN SELECT 1 FROM system.iota; END
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateUser('u5', 'pw5', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g1', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g2', '')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u3', 'g1')
direct GRANT SELECT, INSERT ON t1 TO u2 WITH GRANT
direct GRANT SELECT ON t2 TO u2
direct GRANT ALL ON t4 TO u2 WITH GRANT
direct GRANT SELECT ON vw1 TO u2 WITH GRANT
direct GRANT UPDATE (id) ON t3 TO u2 WITH GRANT
direct GRANT CREATE VIEW TO u2 WITH GRANT
direct GRANT CREATE PROCEDURE TO u2
direct GRANT SELECT ON t1 TO g1 WITH GRANT
direct GRANT SELECT ON t2 TO g1
query SELECT * FROM system.permissions
disconnect
#
# The user holds some rights with the grant option, some without.
connect go\go.add u2 pw2
stmt
direct GRANT SELECT ON t1 TO u5
direct GRANT INSERT ON t1 TO u5
direct GRANT UPDATE ON t1 TO u5
direct GRANT DELETE ON t1 TO g2
direct GRANT SELECT ON t2 TO u5
direct GRANT SELECT ON t3 TO u5
direct GRANT SELECT ON t4 TO u5 WITH GRANT
direct GRANT UPDATE ON t4 TO u5
direct GRANT ALL ON t4 TO g2
direct GRANT ALL ON t1 TO u5
direct GRANT SELECT ON vw1 TO u5
direct GRANT SELECT ON vw2 TO u5
direct GRANT SELECT ON pr1 TO u5
direct GRANT EXECUTE ON pr1 TO u5
direct GRANT UPDATE (id) ON t3 TO u5
direct GRANT UPDATE (id) ON t1 TO u5
direct GRANT SELECT (id) ON t1 TO u5
direct GRANT SELECT ON t1 TO u2
direct GRANT CREATE VIEW TO u5
direct GRANT CREATE PROCEDURE TO u5
direct GRANT CREATE TABLE TO u5
direct GRANT CREATE VIEW TO g2
direct GRANT UPDATE ON nosuch TO u5
direct GRANT UPDATE ON t3 TO nosuch
direct REVOKE SELECT ON t1 FROM u5
direct REVOKE UPDATE ON t1 FROM u5
direct REVOKE INSERT ON t2 FROM u5
direct REVOKE SELECT ON t2 FROM u5
direct REVOKE CREATE VIEW FROM u5
direct REVOKE CREATE TABLE FROM u5
direct REVOKE ALL ON t4 FROM g2
direct REVOKE UPDATE (id) ON t3 FROM u5
direct GRANT ALL ON t2 TO u5
direct GRANT ALL ON t3 TO u5
direct GRANT ALL ON vw1 TO u5
direct GRANT SELECT ON t1 TO u5 WITH GRANT
direct GRANT UPDATE ON t1 TO u5 WITH GRANT
direct GRANT SELECT, UPDATE ON t4 TO g2
direct GRANT UPDATE, SELECT ON t1 TO g2
direct REVOKE GRANT OPTION FOR SELECT ON t1 FROM u5
direct REVOKE GRANT OPTION FOR UPDATE ON t1 FROM u5
direct REVOKE ALL ON t1 FROM u5
direct REVOKE ALL ON t2 FROM u5
direct REVOKE SELECT ON t3 FROM u5
direct GRANT CREATE VIEW TO u5 WITH GRANT
direct GRANT CREATE TABLE TO u5 WITH GRANT
direct GRANT SELECT ON t4 TO u2
disconnect
connect go\go.add adssys
stmt
query SELECT * FROM system.permissions
disconnect
#
# A member of a group that holds the grant option.
connect go\go.add u3 pw3
stmt
direct GRANT SELECT ON t1 TO u5
direct GRANT INSERT ON t1 TO u5
direct GRANT SELECT ON t2 TO u5
direct GRANT SELECT ON t4 TO u5
disconnect
#
# What it was given, it may give on; the rights it makes itself.
connect go\go.add adssys
stmt
direct GRANT CREATE TABLE TO u5 WITH GRANT
direct GRANT SELECT ON t3 TO u5
direct REVOKE SELECT ON t4 FROM u2
disconnect
connect go\go.add u5 pw5
stmt
direct GRANT SELECT ON t3 TO u3
direct CREATE TABLE n1 (id INTEGER) IN DATABASE
direct GRANT SELECT ON n1 TO u3
direct GRANT SELECT ON n1 TO u3 WITH GRANT
direct GRANT CREATE TABLE TO u3
direct GRANT CREATE VIEW TO u3
disconnect
connect go\go.add u2 pw2
stmt
direct GRANT SELECT ON t4 TO u5
direct GRANT SELECT ON t1 TO u3 WITH GRANT
disconnect
connect go\go.add u3 pw3
stmt
direct GRANT SELECT ON t1 TO u5
direct GRANT INSERT ON t1 TO u5
disconnect
connect go\go.add adssys
stmt
query SELECT * FROM system.permissions
disconnect
#
# ALL: what a user holding some of the rights, or none, gives with it.
connect go\go.add u2 pw2
stmt
direct GRANT ALL ON t2 TO g2
direct GRANT ALL ON t3 TO g2
direct GRANT ALL ON t1 TO g2
direct GRANT ALL ON vw1 TO g2
disconnect
connect go\go.add u5 pw5
stmt
direct GRANT ALL ON t1 TO u3
direct GRANT ALL ON vw2 TO u3
direct GRANT ALL ON n1 TO u3
disconnect
connect go\go.add adssys
stmt
query SELECT * FROM system.permissions
