# What ddopts.txt does, on a dictionary connection (ct\ct.add), in an empty
# folder: CREATE INDEX options (IN FILE puts an index in a file of the
# dictionary), IGNORE and RESPECT TRANSACTIONS, the drops of ALTER TABLE
# (the table's primary key, a column's constraints), DROP TABLE ... FROM
# DATABASE [NO_DELETE], and the same statements on a free connection to the
# dictionary's folder. The dictionary and the table files it leaves are kept
# in fixtures/sql/ddoptsdict/:
#
#     run.sh <work> <ace32.dll> ddoptsdict.txt > ../../../fixtures/sql/ddoptsdict.json
#     cp <work>/data/ct/* ../../../fixtures/sql/ddoptsdict/
ddcreate ct\ct.add
