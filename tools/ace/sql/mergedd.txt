# MERGE on a dictionary connection: the dictionary's defaults, field constraints,
# primary key, AutoInc column and triggers, in an empty folder with the
# dictionary the script creates, ct/ct.add; the files it leaves are kept in
# fixtures/sql/mergedd/:
#
#     run.sh <work> <ace32.dll> mergedd.txt > ../../../fixtures/sql/mergedd.json
#     cp <work>/data/ct/* ../../../fixtures/sql/mergedd/
ddcreate ct\ct.add
stmt
direct CREATE TABLE acct (id INTEGER CONSTRAINT NOT NULL, name CHAR(10) DEFAULT 'anon', qty INTEGER CONSTRAINT MINIMUM '0' CONSTRAINT MAXIMUM '100' CONSTRAINT ERROR MESSAGE 'qty out of range' DEFAULT '5', a AUTOINC, PRIMARY KEY (id))
direct CREATE TABLE log (msg CHAR(60))
direct CREATE TABLE feed (id INTEGER, name CHAR(10), qty INTEGER)

direct INSERT INTO acct DEFAULT VALUES

# Triggers, in the order they fire
direct CREATE TRIGGER a_bi ON acct BEFORE INSERT BEGIN INSERT INTO log (msg) SELECT 'bi ' + TRIM(CAST(id AS SQL_CHAR)) FROM __new; END
direct CREATE TRIGGER a_ai ON acct AFTER INSERT BEGIN INSERT INTO log (msg) SELECT 'ai ' + TRIM(CAST(id AS SQL_CHAR)) FROM __new; END
direct CREATE TRIGGER a_bu ON acct BEFORE UPDATE BEGIN INSERT INTO log (msg) SELECT 'bu ' + TRIM(CAST(o.qty AS SQL_CHAR)) + '>' + TRIM(CAST(n.qty AS SQL_CHAR)) FROM __old o, __new n; END
direct CREATE TRIGGER a_au ON acct AFTER UPDATE BEGIN INSERT INTO log (msg) SELECT 'au ' + TRIM(CAST(o.qty AS SQL_CHAR)) + '>' + TRIM(CAST(n.qty AS SQL_CHAR)) FROM __old o, __new n; END

# An INSTEAD OF trigger takes the place of the insert and of the update
direct CREATE TABLE inst (id INTEGER, qty INTEGER)
direct INSERT INTO inst VALUES (1, 1)
direct CREATE TRIGGER i_ii ON inst INSTEAD OF INSERT BEGIN INSERT INTO log (msg) SELECT 'ii ' + TRIM(CAST(id AS SQL_CHAR)) FROM __new; END
direct CREATE TRIGGER i_iu ON inst INSTEAD OF UPDATE BEGIN INSERT INTO log (msg) SELECT 'iu ' + TRIM(CAST(o.qty AS SQL_CHAR)) + '>' + TRIM(CAST(n.qty AS SQL_CHAR)) FROM __old o, __new n; END
direct MERGE inst i USING feed f ON i.id = f.id WHEN MATCHED THEN UPDATE SET qty = f.qty WHEN NOT MATCHED THEN INSERT (id, qty) VALUES (f.id, f.qty)
