# A trigger's savepoints, on the remote server (SQL_SERVER_TYPE=2, see triggers.txt),
# in a dictionary of an empty folder, pq/pq.add (txtrig.txt has the rest of the
# triggers' transactions): an AFTER INSERT trigger that rolls back to a savepoint
# it set takes back the insert that fired it, and what it wrote after the
# savepoint; a NO TRANSACTION trigger is a call as a procedure is, and may not roll
# back to the caller's savepoints (7129) nor end the caller's levels (5047). A
# statement of its own (stmt) comes before each CREATE TRIGGER: the original goes on
# with the triggers a table had when it was opened, which the statement holds.
#
#     SQL_SERVER_TYPE=2 run.sh <work> <ace32.dll> txtrig2.txt <share>
#             > ../../../fixtures/sql/txtrig2.json
stmt
direct CREATE DATABASE "pq\pq.add"
connect pq\pq.add ADSSYS
stmt
direct CREATE TABLE t (id INTEGER, name CHAR(10))
direct CREATE TABLE u (id INTEGER)
direct CREATE TABLE w (id INTEGER)
direct CREATE TABLE audit (msg CHAR(60)) IGNORE TRANSACTIONS
#
# AFTER INSERT: a savepoint, a write, a rollback to the savepoint, outside a
# transaction and in one; what the trigger wrote before the savepoint stays.
closestmt
stmt
direct CREATE TRIGGER s1 ON t AFTER INSERT BEGIN SAVEPOINT inner1; INSERT INTO u VALUES (5); ROLLBACK TO SAVEPOINT inner1; END
direct INSERT INTO t VALUES (1, 'a')
txcount
query SELECT id FROM t
query SELECT id FROM u
open t.adt
direct BEGIN TRANSACTION
direct INSERT INTO t VALUES (2, 'b')
txcount
query SELECT id FROM t
query SELECT id FROM u
direct COMMIT
query SELECT id FROM t
direct DROP TRIGGER t.s1
direct DELETE FROM t
direct DELETE FROM u
closestmt
stmt
direct CREATE TRIGGER s2 ON t AFTER INSERT BEGIN INSERT INTO u VALUES (6); SAVEPOINT inner2; INSERT INTO u VALUES (7); ROLLBACK TO SAVEPOINT inner2; END
direct INSERT INTO t VALUES (3, 'c')

query SELECT id FROM u
direct DROP TRIGGER t.s2
direct DELETE FROM t
direct DELETE FROM u
