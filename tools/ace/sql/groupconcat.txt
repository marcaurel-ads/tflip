# Default comma separator; CHAR values keep their padding.
query SELECT grp, GROUP_CONCAT(txt) AS g FROM gc GROUP BY grp ORDER BY grp
# A VARCHAR argument is not padded.
query SELECT grp, GROUP_CONCAT(vtxt) AS g FROM gc GROUP BY grp ORDER BY grp
# A memo argument is rejected, like a numeric one.
query SELECT grp, GROUP_CONCAT(m) AS g FROM gc GROUP BY grp ORDER BY grp
# An explicit separator, and an empty one.
query SELECT grp, GROUP_CONCAT(txt SEPARATOR '; ') AS g FROM gc GROUP BY grp ORDER BY grp
query SELECT grp, GROUP_CONCAT(txt SEPARATOR '') AS g FROM gc GROUP BY grp ORDER BY grp
# DISTINCT keeps the first occurrence; with ORDER BY it sorts.
query SELECT grp, GROUP_CONCAT(DISTINCT txt) AS g FROM gc GROUP BY grp ORDER BY grp
query SELECT grp, GROUP_CONCAT(DISTINCT txt ORDER BY txt) AS g FROM gc GROUP BY grp ORDER BY grp
query SELECT grp, GROUP_CONCAT(txt ORDER BY txt DESC) AS g FROM gc GROUP BY grp ORDER BY grp
# An expression argument.
query SELECT grp, GROUP_CONCAT(UPPER(txt) SEPARATOR '|') AS g FROM gc GROUP BY grp ORDER BY grp
# NULLs are skipped; an empty group is NULL.
query SELECT GROUP_CONCAT(txt) AS g FROM gc WHERE grp = 99
# A numeric argument is rejected.
query SELECT GROUP_CONCAT(num) AS g FROM gc
