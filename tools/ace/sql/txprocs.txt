#
# The count a procedure reads at each step, logged in a table the transactions
# leave alone, called at the top level and in the caller's transaction.
direct CREATE TABLE lg (i AUTOINC, k CHAR(20), c INTEGER) IGNORE TRANSACTIONS
direct CREATE PROCEDURE q1(a INTEGER) BEGIN INSERT INTO lg (k, c) SELECT 'entry', ::conn.TransactionCount FROM system.iota; BEGIN TRANSACTION; INSERT INTO lg (k, c) SELECT 'after begin', ::conn.TransactionCount FROM system.iota; INSERT INTO t VALUES (_a, 'q1'); COMMIT; INSERT INTO lg (k, c) SELECT 'after commit', ::conn.TransactionCount FROM system.iota; END
direct CREATE PROCEDURE q2(a INTEGER) BEGIN INSERT INTO lg (k, c) SELECT 'entry', ::conn.TransactionCount FROM system.iota; BEGIN TRANSACTION; INSERT INTO lg (k, c) SELECT 'after begin', ::conn.TransactionCount FROM system.iota; INSERT INTO t VALUES (_a, 'q2'); TRY ROLLBACK; CATCH ALL INSERT INTO lg (k, c) SELECT 'rollback err', __errcode FROM system.iota; END TRY; INSERT INTO lg (k, c) SELECT 'after rollback', ::conn.TransactionCount FROM system.iota; END
direct CREATE PROCEDURE q3(a INTEGER) BEGIN INSERT INTO lg (k, c) SELECT 'entry', ::conn.TransactionCount FROM system.iota; TRY COMMIT; CATCH ALL INSERT INTO lg (k, c) SELECT 'commit err', __errcode FROM system.iota; END TRY; TRY ROLLBACK; CATCH ALL INSERT INTO lg (k, c) SELECT 'rollback err', __errcode FROM system.iota; END TRY; INSERT INTO lg (k, c) SELECT 'exit', ::conn.TransactionCount FROM system.iota; END
direct DELETE FROM t
direct EXECUTE PROCEDURE q1(1)
