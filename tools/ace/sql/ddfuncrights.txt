# A rename of a function by a user who is not the administrator: whether
# the right to alter it, the right to create functions, or the name is
# checked first, and the same through AdsDDRenameObject (functions,
# packages and the objects of the other types). The dictionary,
# fr/fr.add, is kept in fixtures/sql/ddfuncrights/:
#
#     run.sh <work> <ace32.dll> ddfuncrights.txt > ../../../fixtures/sql/ddfuncrights.json
#     cp <work>/data/fr/* ../../../fixtures/sql/ddfuncrights/
ddcreate fr\fr.add
stmt
direct CREATE TABLE t1 (id INTEGER, name CHAR(10))
direct CREATE FUNCTION fa(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION fb(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION fc(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END

direct CREATE PACKAGE pa
direct CREATE PACKAGE pb
direct CREATE PACKAGE pc
direct CREATE FUNCTION pa.x(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE FUNCTION pa.y(a INTEGER) RETURNS INTEGER BEGIN RETURN a; END
direct CREATE PROCEDURE pr1() BEGIN SELECT 1 FROM system.iota; END
direct CREATE VIEW vw1 AS SELECT id FROM t1
direct CREATE TABLE t9 (id INTEGER)
direct CREATE VIEW vw9 AS SELECT id FROM t1
direct CREATE PROCEDURE pr9() BEGIN SELECT 1 FROM system.iota; END
direct EXECUTE PROCEDURE sp_CreateLink('lk1', 'fr.add', 1, 0, 0, 'ADSSYS', '')
direct EXECUTE PROCEDURE sp_CreateLink('lk2', 'fr.add', 1, 0, 0, 'ADSSYS', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g1', '')
direct EXECUTE PROCEDURE sp_CreateGroup('g2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_CreateUser('u4', 'pw4', '')
direct GRANT ALTER ON fa TO u2
direct GRANT ALTER ON pa TO u2
direct GRANT ALTER ON fc TO u3
direct GRANT CREATE FUNCTION TO u3
direct GRANT CREATE PACKAGE TO u3
direct GRANT ALTER ON pr1 TO u2
direct GRANT ALTER ON t1 TO u2
direct GRANT ALTER ON vw1 TO u2
direct GRANT ALTER ON lk1 TO u2
direct GRANT ALTER ON g1 TO u2
direct GRANT CREATE FUNCTION TO u4
direct GRANT CREATE PACKAGE TO u4
disconnect
