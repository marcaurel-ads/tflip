# Membership in the built-in groups (DB:Admin, DB:Backup, DB:Debug,
# SERVER:Admin, SERVER:Monitor), which sp_AddUserToGroup gives and
# sp_RemoveUserFromGroup takes, and what a member of DB:Admin may do. The
# dictionary, bg/bg.add, is kept in fixtures/sql/ddbuiltin/:
#
#     run.sh <work> <ace32.dll> ddbuiltin.txt > ../../../fixtures/sql/ddbuiltin.json
#     cp <work>/data/bg/* ../../../fixtures/sql/ddbuiltin/

direct EXECUTE PROCEDURE sp_CreateUser('u4', 'pw4', '')
direct EXECUTE PROCEDURE sp_CreateUser('u5', 'pw5', '')

#
# A member of DB:Admin holds every right; one of DB:Backup, DB:Debug and the
# SERVER groups none.
connect bg\bg.add u1 pw1
stmt
direct CREATE VIEW vw1 AS SELECT id FROM t1
direct CREATE PROCEDURE pr1() BEGIN SELECT 1 FROM system.iota; END
direct GRANT SELECT ON t1 TO u2 WITH GRANT
direct EXECUTE PROCEDURE sp_CreateUser('u7', 'pw7', '')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u7', 'DB:Admin')
direct DROP TABLE t2
query SELECT * FROM t1
disconnect
connect bg\bg.add u2 pw2
stmt

direct GRANT SELECT ON t1 TO u3
direct EXECUTE PROCEDURE sp_AddUserToGroup('u3', 'g1')

disconnect
