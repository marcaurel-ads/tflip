# SQL comparisons of tables opened with a dynamic collation
# (AdsStmtSetTableCollation), under the default static collation
# (run_collation_dynamic.sh): the charset_german `ca` table (c is each byte
# 32..255, w holds accented words; no index) and the collation_dynamic `ger`
# and `gen` tables (the same bytes and more words, with an order under
# GERMAN_VFP_CI_AS_1252 and GENERAL_VFP_CI_AS_1252).
#
# A table whose orders have a collation opens only with that one, so the
# tables of one statement cannot differ.
stmt
query SELECT code FROM ger ORDER BY code
collation GENERAL_VFP_CI_AS_1252
query SELECT code FROM ger ORDER BY code
query SELECT a.code FROM ca a, gen b WHERE a.code = b.code ORDER BY a.code
query SELECT a.code FROM ca a, ger b WHERE a.code = b.code ORDER BY a.code
