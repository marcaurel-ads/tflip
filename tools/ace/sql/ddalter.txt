# ALTER TABLE of a dictionary's tables that changes fields: ADD, ALTER and
# DROP COLUMN with the constraints of a column, on a dictionary connection
# (ct\ct.add) in an empty folder. A change that only names fields again
# renames them where they are; any other restructures the table (the old files
# left as .BAK) and makes the table's Field objects anew. The dictionary and
# the files it leaves are kept in fixtures/sql/ddalter/:
#
#     run.sh <work> <ace32.dll> ddalter.txt > ../../../fixtures/sql/ddalter.json
#     cp <work>/data/ct/* ../../../fixtures/sql/ddalter/
ddcreate ct\ct.add
stmt
dduser u1 pw
#
# ADD COLUMN, with the constraints of a column, at a position, with a column
# that cannot be NULL on a table with records.
direct CREATE TABLE ta (a INTEGER CONSTRAINT NOT NULL CONSTRAINT MINIMUM '1' CONSTRAINT MAXIMUM '9' CONSTRAINT ERROR MESSAGE 'bad' DEFAULT '3', b CHAR(4) DEFAULT 'x', c CHAR(3), PRIMARY KEY (a))
direct CREATE INDEX tab ON ta (b)
direct INSERT INTO ta VALUES (1, 'x', 'p')
direct INSERT INTO ta VALUES (2, 'y', 'q')
direct ALTER TABLE ta ADD COLUMN d CHAR(5) DEFAULT 'abc' CONSTRAINT MINIMUM 'A' CONSTRAINT MAXIMUM 'z' CONSTRAINT NOT NULL CONSTRAINT ERROR MESSAGE 'dbad'
direct ALTER TABLE ta ADD COLUMN e INTEGER POSITION 1
direct ALTER TABLE ta ADD COLUMN f INTEGER CONSTRAINT NOT NULL
direct ALTER TABLE ta ADD COLUMN g INTEGER DEFAULT 'abc' CONSTRAINT MINIMUM 'abc'
direct ALTER TABLE ta ADD COLUMN h DOUBLE ADD COLUMN i CHAR(7) DEFAULT 'q' ADD COLUMN j MEMO
query SELECT Name, Parent, Field_Num, Field_Type, Field_Length, Field_Can_Be_Null, Field_Default_Value, Field_Min_Value, Field_Max_Value, Field_Validation_Msg FROM system.columns WHERE Parent = 'ta'
query SELECT * FROM ta
