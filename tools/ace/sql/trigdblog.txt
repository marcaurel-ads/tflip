# Database triggers on OPEN_TABLE and CLOSE_TABLE: statements.
#
# What a statement opens, when it closes it: DML, DDL, SELECTs of one table or
# several, the cursor the client opens of a live SELECT, the same SQL run again
# (which keeps its tables), another statement (which closes them first), a
# statement closed with a cursor open, the order a statement closes several.
#
# Database triggers ON DATABASE BEFORE | AFTER OPEN_TABLE and CLOSE_TABLE that
# log what __info tells them (the table's name, the number of times it is open,
# whether it is shared, free, which type, and whether a client opened it) to a
# table logt, in an empty folder with a dictionary dbt/dbt.add the script
# creates (kept in fixtures/sql/trigdblog/, which the replay compares but for logt
# and its memo file). Recorded on the local server with no statements' tables
# kept open (AdsCacheOpenCursors 0), and replayed: what the triggers write is
# read back at the end, by the statements' own results:
#
#     run.sh <work> <ace32.dll> trigdblog.txt > ../../../fixtures/sql/trigdblog.json
#     cp <work>/data/dbt/* ../../../fixtures/sql/trigdblog/
#
# A marker statement in logt says where each case ends. The triggers leave the
# opens of logt alone (their own writes open it too, which fire none of the
# kind they run in).
cachecursors 0
ddcreate dbt\dbt.add
stmt
direct CREATE TABLE logt (n AUTOINC, msg VARCHAR(200))
direct CREATE TABLE t1 (id INTEGER)
direct CREATE TABLE t2 (id INTEGER)
direct INSERT INTO t1 VALUES (1)
direct INSERT INTO t2 VALUES (1)
direct CREATE TRIGGER a1 ON DATABASE BEFORE OPEN_TABLE BEGIN INSERT INTO logt (msg) SELECT 'B-OP ' + IFNULL(TRIM(TableName),'N') + ' ' + IFNULL(CAST(UserID AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(OpenCount AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsShared AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsFree AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(LockType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(TableType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsDirect AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsTemp AS SQL_CHAR(10)),'N') FROM __info WHERE IFNULL(TRIM(TableName),'') <> 'logt'; END
direct CREATE TRIGGER a2 ON DATABASE AFTER OPEN_TABLE BEGIN INSERT INTO logt (msg) SELECT 'A-OP ' + IFNULL(TRIM(TableName),'N') + ' ' + IFNULL(CAST(UserID AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(OpenCount AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsShared AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsFree AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(LockType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(TableType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsDirect AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsTemp AS SQL_CHAR(10)),'N') FROM __info WHERE IFNULL(TRIM(TableName),'') <> 'logt'; END
direct CREATE TRIGGER a3 ON DATABASE BEFORE CLOSE_TABLE BEGIN INSERT INTO logt (msg) SELECT 'B-CL ' + IFNULL(TRIM(TableName),'N') + ' ' + IFNULL(CAST(UserID AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(OpenCount AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsShared AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsFree AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(LockType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(TableType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsDirect AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsTemp AS SQL_CHAR(10)),'N') FROM __info WHERE IFNULL(TRIM(TableName),'') <> 'logt'; END
direct CREATE TRIGGER a4 ON DATABASE AFTER CLOSE_TABLE BEGIN INSERT INTO logt (msg) SELECT 'A-CL ' + IFNULL(TRIM(TableName),'N') + ' ' + IFNULL(CAST(UserID AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(OpenCount AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsShared AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsFree AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(LockType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(TableType AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsDirect AS SQL_CHAR(10)),'N') + ' ' + IFNULL(CAST(IsTemp AS SQL_CHAR(10)),'N') FROM __info WHERE IFNULL(TRIM(TableName),'') <> 'logt'; END
direct DELETE FROM logt
