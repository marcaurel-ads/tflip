# What a member of DB:Admin, and a user holding the right to create tables,
# gets from a CREATE TABLE with a PRIMARY KEY, UNIQUE and other constraints,
# and from the statements that add them. The dictionary, ap/ap.add, is kept
# in fixtures/sql/ddadminpk/:
#
#     run.sh <work> <ace32.dll> ddadminpk.txt > ../../../fixtures/sql/ddadminpk.json
#     cp <work>/data/ap/* ../../../fixtures/sql/ddadminpk/
ddcreate ap\ap.add
stmt
direct CREATE TABLE t0 (id INTEGER PRIMARY KEY, name CHAR(10))
direct EXECUTE PROCEDURE sp_CreateUser('u1', 'pw1', '')
direct EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
direct EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
direct EXECUTE PROCEDURE sp_AddUserToGroup('u1', 'DB:Admin')
direct GRANT CREATE TABLE TO u2
direct GRANT CREATE TABLE TO u3
direct GRANT CREATE VIEW TO u3
disconnect
#
# A member of DB:Admin.
connect ap\ap.add u1 pw1
stmt
direct CREATE TABLE a1 (id INTEGER PRIMARY KEY, name CHAR(10))
direct CREATE TABLE a2 (id INTEGER, name CHAR(10), PRIMARY KEY (id))
direct CREATE TABLE a3 (id INTEGER, name CHAR(10) NOT NULL)
direct CREATE TABLE a4 (id INTEGER, name CHAR(10), UNIQUE (name))
direct CREATE TABLE a5 (id INTEGER, name CHAR(10) UNIQUE)
direct CREATE TABLE a6 (id INTEGER NOT NULL, name CHAR(10), PRIMARY KEY (id))
direct CREATE TABLE a7 (id INTEGER, ref INTEGER, FOREIGN KEY (ref) REFERENCES t0 (id))
direct CREATE TABLE a8 (id INTEGER PRIMARY KEY, name CHAR(10)) IN DATABASE
direct CREATE TABLE a9 (id INTEGER, name CHAR(10))
direct CREATE TABLE a10 (id INTEGER, name CHAR(10)) IN DATABASE
direct CREATE TABLE a11 (id INTEGER NOT NULL) AS FREE TABLE
direct CREATE TABLE a12 (id INTEGER PRIMARY KEY) AS FREE TABLE
direct ALTER TABLE a9 ADD PRIMARY KEY (id)
direct ALTER TABLE a9 ADD COLUMN x INTEGER
direct CREATE INDEX i9 ON a9 (name)
direct CREATE UNIQUE INDEX i9u ON a9 (x)
query SELECT * FROM system.tables
query SELECT * FROM system.indexes
disconnect
#
# A user with the right to create tables.
connect ap\ap.add u2 pw2
stmt
direct CREATE TABLE b1 (id INTEGER PRIMARY KEY, name CHAR(10)) IN DATABASE
direct CREATE TABLE b2 (id INTEGER, name CHAR(10), PRIMARY KEY (id)) IN DATABASE
direct CREATE TABLE b3 (id INTEGER, name CHAR(10)) IN DATABASE
direct CREATE TABLE b4 (id INTEGER PRIMARY KEY, name CHAR(10))
direct CREATE TABLE b5 (id INTEGER, name CHAR(10) UNIQUE) IN DATABASE
direct CREATE TABLE b6 (id INTEGER, name CHAR(10))
direct CREATE TABLE b7 (id INTEGER, name CHAR(10)) AS FREE TABLE
disconnect
connect ap\ap.add u3 pw3
stmt
direct CREATE TABLE c1 (id INTEGER PRIMARY KEY, name CHAR(10)) IN DATABASE
direct CREATE VIEW c2 AS SELECT id FROM c1
direct CREATE TABLE c3 (id INTEGER)
direct CREATE TABLE c4 (id INTEGER, name CHAR(10) NOT NULL) 
disconnect
connect ap\ap.add adssys
stmt
query SELECT * FROM system.tables
query SELECT * FROM system.indexes
query SELECT * FROM system.permissions
disconnect
