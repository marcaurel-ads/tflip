# Trigger libraries (aeptrig.c as aeptrig.aep), on the remote server
# (SQL_SERVER_TYPE=2): run.sh builds the library into the folder ae/ of the
# data folder, in which this script makes its dictionary. The arguments the
# library gets for each event and timing, __new and __old read through the
# handle, a write to __new, what the change shows to the library, the
# library's writes with the change's transaction, the failures (__error with
# and without a number, a non-zero return, a statement that fails, a function
# and a library that are not there), Startup and Shutdown, NO VALUES and
# NO MEMOS, and what a library does as a database trigger.
#
#     tools/ace/aep/run.sh aeptrig.txt aeptrig.c aeptrig.aep > fixtures/sql/aeptrig.json
stmt
direct CREATE DATABASE "ae\ae.add"
connect ae\ae.add ADSSYS
stmt
direct CREATE TABLE log (what VARCHAR(20), a VARCHAR(80), b INTEGER, c INTEGER, d INTEGER, seq AUTOINC)
#
# The arguments, by event and timing
direct CREATE TABLE t1 (n INTEGER, memo MEMO)
direct CREATE TRIGGER t1_ib ON t1 BEFORE INSERT FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 1
direct CREATE TRIGGER t1_ia ON t1 AFTER INSERT FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 2
direct CREATE TRIGGER t1_ub ON t1 BEFORE UPDATE FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 1
direct CREATE TRIGGER t1_ua ON t1 AFTER UPDATE FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 2
direct CREATE TRIGGER t1_db ON t1 BEFORE DELETE FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 1
direct CREATE TRIGGER t1_da ON t1 AFTER DELETE FUNCTION Args IN LIBRARY "aeptrig.aep" PRIORITY 2
query SELECT Name, Trig_Event_Type, Trig_Trigger_Type, Trig_Container_Type, Trig_Container, Trig_Function_Name, Trig_Priority, Trig_Options FROM system.triggers
query INSERT INTO t1 (n) VALUES (1)
query INSERT INTO t1 (n) VALUES (2), (3)
query INSERT INTO t1 (n) SELECT n + 10 FROM t1
query UPDATE t1 SET n = n + 1 WHERE n < 4
query DELETE FROM t1 WHERE n > 10
query SELECT n FROM t1
query SELECT what, a, b, c, d FROM log ORDER BY seq
direct DELETE FROM log
