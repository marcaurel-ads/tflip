# Steps for freetables.c (see there), on a copy of fixtures/adt/dictionary_props
# with the free tables indexed.adt (ADT) and dbf_types.dbf (CDX) beside it. The
# dictionary has ADSSYS (adssyspw) and bob (bobpw, in group readers, with
# SELECT and INSERT on parent and UPDATE(label), nothing on child), and parent
# has TABLE_AUTO_CREATE on.
#
# 1. Free tables on a dictionary connection, with and without the connection
#    option ADS_CHECK_FREE_TABLE_ACCESS (0x2000, an AdsConnect60 option).
conn admin dict ADSSYS adssyspw 0
conn adminchk dict ADSSYS adssyspw 0x2000
conn bob dict bob bobpw 0
conn bobchk dict bob bobpw 0x2000
conn free free - - 0
open admin indexed.adt 3 0
close indexed.adt
open bob indexed.adt 3 0
close indexed.adt
open adminchk indexed.adt 3 0
close indexed.adt
open bobchk indexed.adt 3 0
open bobchk indexed.adt 3 2
open bob dbf_types.dbf 2 0
close dbf_types.dbf
open bobchk dbf_types.dbf 2 0
open adminchk dbf_types.dbf 2 0
close dbf_types.dbf
open free indexed.adt 3 0
close indexed.adt
open bob indexed.adt 3 0x1000
sql bob SELECT * FROM indexed
sql bobchk SELECT * FROM indexed
sql adminchk SELECT * FROM indexed
sql bobchk SELECT * FROM "indexed.adt"
sql bobchk SELECT * FROM parent
sql bob CREATE TABLE f1 (id integer) AS FREE TABLE
exists f1.adt
sql bob INSERT INTO f1 VALUES (1)
sql bob UPDATE f1 SET id = 2
sql bob DELETE FROM f1
sql bobchk CREATE TABLE f2 (id integer) AS FREE TABLE
exists f2.adt
sql bobchk INSERT INTO f1 VALUES (3)
sql bobchk UPDATE f1 SET id = 4
sql bobchk DELETE FROM f1
sql bobchk SELECT * FROM f1
sql bobchk ALTER TABLE f1 ADD COLUMN x integer
sql bobchk DROP TABLE f1
exists f1.adt
sql bob ALTER TABLE f1 ADD COLUMN x integer
sql bob DROP TABLE f1
exists f1.adt
sql adminchk CREATE TABLE f3 (id integer) AS FREE TABLE
exists f3.adt
sql adminchk INSERT INTO f3 VALUES (1)
sql adminchk DROP TABLE f3
exists f3.adt
disc bobchk
disc adminchk
disc free
