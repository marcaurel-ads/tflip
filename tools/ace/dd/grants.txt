# The scenarios grants.exe runs (see grants.c for the verbs). Each `@ name`
# starts a new dictionary; grants.sh keeps the ones named there as goldens.

# SQL GRANT on every kind of object, found by its bare name.
@ sqlgrant
user bob
group G1
user bob2
group G2
table t
view v SELECT id FROM t
proc p
link lnk
pub pb
sql GRANT SELECT ON v TO bob
sql GRANT EXECUTE ON p TO bob
sql GRANT ACCESS ON lnk TO bob
sql GRANT ALTER, DROP ON lnk TO G1
sql GRANT ALTER ON bob2 TO bob
sql GRANT DROP ON G2 TO bob
sql GRANT ALL ON pb TO bob
sql GRANT ALL ON lnk TO G2 WITH GRANT
sql GRANT INSERT(s), UPDATE(s), SELECT(id) ON t TO bob
sql GRANT SELECT(s), DELETE ON t TO G1
sql GRANT ALTER DATABASE TO G2 WITH GRANT
sql GRANT ALL ON [v] TO "bob2"
sql grant execute, inherit on p to bob2 with grant;
sql GRANT ALL ON bob TO G1
sql GRANT INHERIT ON t TO bob2
sql GRANT INHERIT DATABASE TO bob2
sql GRANT SELECT ( s ) ON t TO bob2 -- a comment
get bob|0|t||0
get bob|4|s|t|0
get bob|0|v||0
get G1|0|lnk||0
get bob2|0|p||0
get bob|0|DB:Public||0
get bob|0|nosuch||0
dump SELECT * FROM system.permissions WHERE Grantee IN ('bob', 'G2') AND Object_Type <> 4

# The right to create each kind of object, and its inherit bit.
@ sqlcreate
user bob
group G1
user bob2
group G2

sql GRANT CREATE USER GROUP TO bob WITH GRANT
sql GRANT INHERIT CREATE PROCEDURE TO bob WITH GRANT
sql REVOKE INHERIT CREATE TABLE, CREATE LINK FROM bob
sql GRANT CREATE FUNCTION, CREATE PACKAGE, CREATE PUBLICATION, CREATE SUBSCRIPTION TO G1 WITH GRANT
sql REVOKE GRANT OPTION FOR CREATE PACKAGE FROM G1
sql REVOKE CREATE SUBSCRIPTION FROM G1
sql GRANT CREATE USER, CREATE LINK TO G1
sql REVOKE CREATE LINK FROM G2
sql GRANT INHERIT CREATE VIEW TO bob2
sql REVOKE GRANT OPTION FOR INHERIT CREATE PROCEDURE FROM bob
member G1 bob
dump SELECT * FROM system.permissions WHERE "Create" IS NOT NULL AND Grantee IN ('bob', 'G1', 'bob2', 'G2', 'DB:Admin')
dump SELECT * FROM system.effectivepermissions WHERE "Create" IS NOT NULL AND Grantee IN ('bob', 'G1')

# A failing grant still raises the dictionary's version; a syntax error or a
# permission with no object does not reach it.
@ noraise
user bob
table t
sql GRANT SELECT TO bob
sql GRANT SELECT, CREATE TABLE TO bob
sql GRANT INHERIT, CREATE TABLE TO bob
@ raise
user bob
table t
grant 0|nosuch||bob|0x1

# The per-column rights: a table grant reaches every column, and the

sql REVOKE GRANT OPTION FOR SELECT(s) ON t2 FROM bob

sql GRANT SELECT(s) ON t2 TO G1 WITH GRANT
sql REVOKE SELECT(s) ON t2 FROM G1
sql GRANT SELECT ON t2 TO bob2 WITH GRANT
