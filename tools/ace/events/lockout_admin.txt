# Failed logins on the remote server and ENFORCE_MAX_FAILED_LOGINS: ADSSYS locked out: another administrator and the logins_disabled property of ADSSYS
# The paces and the good logins between the failures keep the server from
# resetting the circuit (error 6624), which it does at times, whatever the
# script: a run that has one is made again.
#
#     EV_SERVER_TYPE=2 ev.exe <ace32.dll> \\127.0.0.1:6262\data\<folder> lockout_admin.txt
#             > ../../../fixtures/events/lockout_admin.json
free 0
sql 0 CREATE DATABASE "d\d.add"
connect 1 d\d.add
sql 1 EXECUTE PROCEDURE sp_CreateUser('a2', 'pw2', '')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('ENFORCE_MAX_FAILED_LOGINS', 'TRUE')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('MAX_FAILED_ATTEMPTS', '2')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('ADMIN_PASSWORD', 'adm')
disconnect 1
connect 1 d\d.add ADSSYS adm
sql 1 EXECUTE PROCEDURE sp_AddUserToGroup('a2', 'DB:Admin')
pace 2000
connect 5 d\d.add ADSSYS adm
disconnect 5
connect 2 d\d.add ADSSYS bad
pace 2000
connect 5 d\d.add ADSSYS adm
disconnect 5
connect 2 d\d.add ADSSYS bad
connect 6 d\d.add a2 pw2
sql 6 EXECUTE PROCEDURE sp_ModifyUserProperty('ADSSYS', 'LOGINS_DISABLED', 'FALSE')
sql 6 SELECT Name, Logins_Disabled FROM system.users
sql 6 EXECUTE PROCEDURE sp_ModifyDatabase('MAX_FAILED_ATTEMPTS', '2')
pace 2000
connect 3 d\d.add ADSSYS adm
disconnect 6
disconnect 1
