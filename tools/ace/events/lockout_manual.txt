# Failed logins on the remote server and ENFORCE_MAX_FAILED_LOGINS: a user an administrator disabled: the answer to a wrong password, with the enforcement on and off
# The paces and the good logins between the failures keep the server from
# resetting the circuit (error 6624), which it does at times, whatever the
# script: a run that has one is made again.
#
#     EV_SERVER_TYPE=2 ev.exe <ace32.dll> \\127.0.0.1:6262\data\<folder> lockout_manual.txt
#             > ../../../fixtures/events/lockout_manual.json
free 0
sql 0 CREATE DATABASE "d\d.add"
connect 1 d\d.add
sql 1 EXECUTE PROCEDURE sp_CreateUser('u2', 'pw2', '')
sql 1 EXECUTE PROCEDURE sp_CreateUser('u3', 'pw3', '')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('ENFORCE_MAX_FAILED_LOGINS', 'TRUE')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('MAX_FAILED_ATTEMPTS', '5')
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('ADMIN_PASSWORD', 'adm')
disconnect 1
connect 1 d\d.add ADSSYS adm
sql 1 EXECUTE PROCEDURE sp_ModifyUserProperty('u2', 'LOGINS_DISABLED', 'TRUE')
pace 2000
connect 5 d\d.add ADSSYS adm
disconnect 5
connect 2 d\d.add u2 bad
pace 2000
connect 3 d\d.add u2 pw2
sql 1 EXECUTE PROCEDURE sp_ModifyDatabase('ENFORCE_MAX_FAILED_LOGINS', 'FALSE')
sql 1 EXECUTE PROCEDURE sp_ModifyUserProperty('u3', 'LOGINS_DISABLED', 'TRUE')
pace 2000
connect 5 d\d.add ADSSYS adm
disconnect 5
connect 2 d\d.add u3 bad
sql 1 SELECT Name, Logins_Disabled FROM system.users
disconnect 1
